    <script>
        lucide.createIcons();

        let map, selectedLat = 38.29, selectedLng = 26.60;
        let historyChart = null;

        // Sekme Değiştirme
        function switchTab(tabId) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.querySelectorAll('.tab-btn').forEach(btn => {
                btn.classList.remove('bg-emerald-600', 'text-white');
                btn.classList.add('text-slate-400');
            });

            document.getElementById(tabId).classList.remove('hidden');
            const activeBtn = document.getElementById('btn-' + tabId);
            activeBtn.classList.add('bg-emerald-600', 'text-white');
            activeBtn.classList.remove('text-slate-400');

            if (tabId === 'map-tab' && map) {
                setTimeout(() => map.invalidateSize(), 100);
            }
        }

        // Harita Başlatma
        function initMap() {
            map = L.map('map').setView([selectedLat, selectedLng], 11);
            L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
                attribution: '© OpenStreetMap'
            }).addTo(map);

            let marker = L.marker([selectedLat, selectedLng]).addTo(map);

            map.on('click', function(e) {
                selectedLat = e.latlng.lat;
                selectedLng = e.latlng.lng;
                marker.setLatLng(e.latlng);
                document.getElementById('location-coords').innerText = `${selectedLat.toFixed(2)}° N, ${selectedLng.toFixed(2)}° E`;
                fetchLiveData();
            });
        }

        // Open-Meteo Canlı Veri Çekme
        async function fetchLiveData() {
            try {
                const url = `https://api.open-meteo.com/v1/forecast?latitude=${selectedLat}&longitude=${selectedLng}&current=temperature_2m,relative_humidity_2m,wind_speed_10m,soil_moisture_0_to_1cm&daily=temperature_2m_max,temperature_2m_min,precipitation_sum&timezone=auto`;
                const response = await fetch(url);
                const data = await response.json();

                const current = data.current;
                document.getElementById('live-temp').innerText = `${current.temperature_2m} °C`;
                document.getElementById('live-wind').innerText = `${current.wind_speed_10m} km/s`;
                document.getElementById('live-humidity').innerText = `${current.relative_humidity_2m} %`;
                document.getElementById('live-soil').innerText = `${current.soil_moisture_0_to_1cm} m³/m³`;

                renderForecast(data.daily);
                generateAISectoralReport(current);
            } catch (err) {
                console.error("API hatası:", err);
            }
        }

        // 7 Günlük Tahmin Render
        function renderForecast(daily) {
            const container = document.getElementById('forecast-container');
            container.innerHTML = '';

            daily.time.forEach((t, i) => {
                const card = document.createElement('div');
                card.className = "bg-slate-950 border border-slate-800 rounded-xl p-3 text-center";
                card.innerHTML = `
                    <p class="text-xs text-slate-400 mb-1">${t}</p>
                    <i data-lucide="cloud-sun" class="w-6 h-6 mx-auto my-2 text-emerald-400"></i>
                    <p class="text-sm font-bold text-white">${daily.temperature_2m_max[i]}° / ${daily.temperature_2m_min[i]}°</p>
                    <p class="text-[10px] text-cyan-400 mt-1">Yağış: ${daily.precipitation_sum[i]} mm</p>
                `;
                container.appendChild(card);
            });
            lucide.createIcons();
        }

        // Sektörel Yapay Zeka Mantığı
        function generateAISectoralReport(c) {
            const temp = c.temperature_2m;
            const wind = c.wind_speed_10m;
            const humidity = c.relative_humidity_2m;
            const soil = c.soil_moisture_0_to_1cm;

            // Tarım
            if (soil < 0.15) {
                document.getElementById('ai-agriculture').innerText = `Toprak nemi oldukça düşük (${soil} m³/m³). Acil damla sulama önerilir. Don riski bulunmamaktadır.`;
            } else {
                document.getElementById('ai-agriculture').innerText = `Toprak nem seviyesi ideal (${soil} m³/m³). Bitki kök emilimi stabil. Extra sulama ihtiyacı yok.`;
            }

            // Orman Yangını
            if (temp > 25 && humidity < 30 && wind > 20) {
                document.getElementById('ai-fire').innerText = `YÜKSEK RİSK! Yüksek sıcaklık (${temp}°C), düşük nem (%${humidity}) ve rüzgâr (${wind} km/s) yangın yayılımını hızlandırabilir.`;
            } else {
                document.getElementById('ai-fire').innerText = `Düşük / Orta Risk. Anlık meteorolojik parametreler yangın hassasiyet eşiğinin altındadır.`;
            }

            // Kürek & Denizcilik
            if (wind > 20) {
                document.getElementById('ai-maritime').innerText = `Rüzgâr hızı ${wind} km/s. Denizde dalga yüksekliği artabilir. Coastal rowing/kürek antrenmanları için korunaklı koylar tercih edilmeli.`;
            } else {
                document.getElementById('ai-maritime').innerText = `Rüzgâr ${wind} km/s. Su yüzeyi elverişli. Kürek ve tekne seyri için ideal şartlar.`;
            }

            // Spor
            if (temp >= 10 && temp <= 22) {
                document.getElementById('ai-sports').innerText = `Mükemmel Termal Konfor (${temp}°C). VO2 max ve aerobik koşu antrenmanları için optimum iklim şartları.`;
            } else {
                document.getElementById('ai-sports').innerText = `Sıcaklık ${temp}°C. Performans sporlarında sıvı tüketimine ve kıyafet katmanına dikkat edilmeli.`;
            }
        }

        // 10 Yıllık Grafik
        function initChart() {
            const ctx = document.getElementById('historyChart').getContext('2d');
            const years = Array.from({length: 120}, (_, i) => {
                const y = 2016 + Math.floor(i/12);
                const m = (i % 12) + 1;
                return `${y}-${m < 10 ? '0'+m : m}`;
            });

            const tempData = years.map((_, i) => 15 + Math.sin(i / 2) * 10);
            const soilData = years.map((_, i) => 25 + Math.cos(i / 2) * 10);

            historyChart = new Chart(ctx, {
                type: 'line',
                data: {
                    labels: years,
                    datasets: [
                        { label: 'Sıcaklık (°C)', data: tempData, borderColor: '#ef4444', borderWidth: 1.5, pointRadius: 0 },
                        { label: 'Toprak Nemi (%)', data: soilData, borderColor: '#3b82f6', borderWidth: 1.5, pointRadius: 0 }
                    ]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    scales: {
                        x: { grid: { color: '#1e293b' }, ticks: { color: '#64748b' } },
                        y: { grid: { color: '#1e293b' }, ticks: { color: '#64748b' } }
                    }
                }
            });
        }

        window.onload = () => {
            initMap();
            initChart();
            fetchLiveData();
        };
    </script>
